// ================================================================
// SETUP
// ================================================================

void setup() {

  orientation(PORTRAIT);

  fullScreen(P3D);

  smooth(8);

  // --------------------------------------------------------------
  // CREATE VECTORS
  // --------------------------------------------------------------

  for (int i = 0; i < NUM_POINTS; i++) {

    points[i]       = new PVector();
    targetPoints[i] = new PVector();

    poseStart[i]    = new PVector();
    poseTarget[i]   = new PVector();

    neutralPose[i]  = new PVector();
  }


  // --------------------------------------------------------------
  // CREATE NEUTRAL BODY
  // --------------------------------------------------------------

  buildNeutralPose(neutralPose);


  // --------------------------------------------------------------
  // INITIAL BODY
  // --------------------------------------------------------------

  copyPose(
    neutralPose,
    points
  );

  copyPose(
    neutralPose,
    targetPoints
  );

  copyPose(
    neutralPose,
    poseStart
  );

  copyPose(
    neutralPose,
    poseTarget
  );


  // --------------------------------------------------------------
  // REST LENGTHS
  // --------------------------------------------------------------

  restLength =
    new float[connections.length];


  for (int i = 0;
       i < connections.length;
       i++) {

    int a =
      connections[i][0];

    int b =
      connections[i][1];

    restLength[i] =
      PVector.dist(
        neutralPose[a],
        neutralPose[b]
      );
  }


  // --------------------------------------------------------------
  // INITIAL KAMAE
  // --------------------------------------------------------------

  buildKamaePose(
    poseTarget
  );

  copyPose(
    poseTarget,
    targetPoints
  );

  copyPose(
    poseTarget,
    points
  );

  copyPose(
    poseTarget,
    poseStart
  );

  transition = 1.0;
}

  // ==============================================================

    drawHomeScreen();

  // ==============================================================
  // CATEGORY
  // ==============================================================

  if (screen == CATEGORY_SCREEN) {

    drawCategoryScreen();

    return;
  }


  // ==============================================================
  // TECHNIQUE
  // ==============================================================

  if (screen == TECHNIQUE_SCREEN) {

    updateAnimation();

    solveEntireSkeleton();

    drawTechniqueScreen();

    return;
  }
}

// ================================================================
// ANIMATION UPDATE
// ================================================================

void updateAnimation() {

  if (walking) {

    walkPhase += 0.075;

    buildWalkPose(
      walkPhase,
      poseTarget
    );
  }


  transition += 0.055;

  transition =
    constrain(
      transition,
      0,
      1
    );


  for (int i = 0;
       i < NUM_POINTS;
       i++) {

    targetPoints[i].set(
      poseStart[i]
    );

    targetPoints[i].lerp(
      poseTarget[i],
      transition
    );
  }
}


// ================================================================
// CONSTRAINT SOLVER
// ================================================================

void solveEntireSkeleton() {

  // --------------------------------------------------------------
  // START FROM DESIRED POSE
  // --------------------------------------------------------------

  for (int i = 0;
       i < NUM_POINTS;
       i++) {

    points[i].set(
      targetPoints[i]
    );
  }


  // --------------------------------------------------------------
  // PBD ITERATIONS
  // --------------------------------------------------------------

  final int ITERATIONS = 18;


  for (int iteration = 0;
       iteration < ITERATIONS;
       iteration++) {


    for (int e = 0;
         e < connections.length;
         e++) {


      int ia =
        connections[e][0];

      int ib =
        connections[e][1];


      PVector a =
        points[ia];

      PVector b =
        points[ib];


      float dx =
        b.x - a.x;

      float dy =
        b.y - a.y;

      float dz =
        b.z - a.z;


      float d =
        sqrt(
          dx * dx +
          dy * dy +
          dz * dz
        );


      if (d < 0.0001) {

        d = 0.0001;
      }


      float desired =
        restLength[e];


      float error =
        (d - desired) / d;


      float stiffness =
        0.90;


      float cx =
        dx * error * 0.5 * stiffness;

      float cy =
        dy * error * 0.5 * stiffness;

      float cz =
        dz * error * 0.5 * stiffness;


      boolean anchorA =
        isPelvisAnchor(ia);

      boolean anchorB =
        isPelvisAnchor(ib);


      if (!anchorA &&
          !anchorB) {

        a.x += cx;
        a.y += cy;
        a.z += cz;

        b.x -= cx;
        b.y -= cy;
        b.z -= cz;
      }


      else if (anchorA &&
               !anchorB) {

        b.x -= cx * 2.0;
        b.y -= cy * 2.0;
        b.z -= cz * 2.0;
      }


      else if (!anchorA &&
               anchorB) {

        a.x += cx * 2.0;
        a.y += cy * 2.0;
        a.z += cz * 2.0;
      }
    }


    // ------------------------------------------------------------
    // KEEP PELVIS ROOTS STABLE
    // ------------------------------------------------------------

    points[23].set(
      targetPoints[23]
    );

    points[24].set(
      targetPoints[24]
    );
  }
}


// ================================================================
// PELVIS ANCHOR
// ================================================================

boolean isPelvisAnchor(
  int index
) {

  return (
    index == 23 ||
    index == 24
  );
}


// ================================================================

// ================================================================

) {

    height / 12.0;

  // HEAD
  // --------------------------------------------------------------

    0

  p[1].set(

    0
  );

  p[2].set(
    -0.32 * s,
    -3.24 * s,
    0
  );

  p[3].set(
    -0.45 * s,
    -3.18 * s,
    0
  );


  p[4].set(
    0.20 * s,
    -3.25 * s,
    0
  );

  p[5].set(
    0.32 * s,
    -3.24 * s,
    0
  );

  p[6].set(
    0.45 * s,
    -3.18 * s,
    0
  );


  p[7].set(
    -0.62 * s,
    -3.05 * s,
    0
  );

  p[8].set(
    0.62 * s,
    -3.05 * s,
    0
  );

// ================================================================
// GENERIC PUNCH
// ================================================================

void buildPunchPose(
  PVector[] p,
  boolean left,
  boolean high
) {

  buildKamaePose(p);

  float s =
    height / 12.0;


  float side =
    left ? -1 : 1;


  float z =
    high
    ? 1.75 * s
    : 1.45 * s;


  float y =
    high
    ? -1.25 * s
    : 0.10 * s;


  int shoulder =
    left ? 11 : 12;

  int elbow =
    left ? 13 : 14;

  int wrist =
    left ? 15 : 16;


  p[elbow].set(
    side * 0.35 * s,
    y,
    z * 0.55
  );


  p[wrist].set(
    side * 0.15 * s,
    y - 0.10 * s,
    z
  );


  syncFingers(p);

  syncFeet(p);
}

// ================================================================
// WALKING
// ================================================================

void buildWalkPose(
  float phase,
  PVector[] p
) {

  buildNeutralPose(p);

  float s =
    height / 12.0;


  float left =
    sin(phase);

  float right =
    sin(phase + PI);


  float leftSwing =
    max(
      0,
      left
    );

  float rightSwing =
    max(
      0,
      right
    );


  float bounce =
    -0.12 *
    s *
    abs(
      sin(
        2 * phase
      )
    );


  float sway =
    0.10 *
    s *
    sin(phase);


  // --------------------------------------------------------------
  // HIPS
  // --------------------------------------------------------------

  p[23].set(
    -0.65 * s + sway,
    1.55 * s + bounce,
    0.18 * s * left
  );

  p[24].set(
    0.65 * s + sway,
    1.55 * s + bounce,
    0.18 * s * right
  );


  // --------------------------------------------------------------
  // SHOULDERS
  // --------------------------------------------------------------

  p[11].set(
    -1.20 * s,
    -1.50 * s + bounce * 0.6,
    -0.10 * s * left
  );

  p[12].set(
    1.20 * s,
    -1.50 * s + bounce * 0.6,
    -0.10 * s * right
  );


  // --------------------------------------------------------------
  // HEAD
  // --------------------------------------------------------------

  p[0].set(
    0.04 * s * sin(phase),
    -3.45 * s + bounce * 0.4,
    0
  );


  // --------------------------------------------------------------
  // LEFT LEG
  // --------------------------------------------------------------

  float lk =
    0.32 * s * leftSwing;


  p[25].set(
    -0.65 * s,
    3.55 * s - lk,
    0.95 * s * left
  );

  p[27].set(
    -0.65 * s,
    5.50 * s - 0.08 * s * leftSwing,
    1.55 * s * left
  );


  // --------------------------------------------------------------
  // RIGHT LEG
  // --------------------------------------------------------------

  float rk =
    0.32 * s * rightSwing;


  p[26].set(
    0.65 * s,
    3.55 * s - rk,
    0.95 * s * right
  );

  p[28].set(
    0.65 * s,
    5.50 * s - 0.08 * s * rightSwing,
    1.55 * s * right
  );


  // --------------------------------------------------------------
  // ARMS OPPOSITE LEGS
  // --------------------------------------------------------------

  p[13].set(
    -1.52 * s,
    -0.05 * s,
    -0.65 * s * left
  );

  p[15].set(
    -1.62 * s,
    1.35 * s,
    -1.10 * s * left
  );


  p[14].set(
    1.52 * s,
    -0.05 * s,
    -0.65 * s * right
  );

  p[16].set(
    1.62 * s,
    1.35 * s,
    -1.10 * s * right
  );


  syncFingers(p);

  syncFeet(p);
}


// ================================================================
// BUILD SELECTED TECHNIQUE
// ================================================================

void buildSelectedTechnique() {

  int c =
    selectedCategory;

  int t =
    selectedTechnique;


  walking = false;


  // ============================================================
  // KIHON
  // ============================================================

  if (c == 0) {

    if (t == 0) {
      buildPunchPose(
        poseTarget,
        false,
        false
      );
    }

    else if (t == 1) {
      buildPunchPose(
        poseTarget,
        true,
        false
      );
    }

    else if (t == 2) {
      buildPunchPose(
        poseTarget,
        true,
        true
      );
    }

    else if (t == 3) {
      buildPunchPose(
        poseTarget,
        false,
        false
      );
    }

    else if (t == 4) {
      buildBlockPose(
        poseTarget,
        0
      );
    }

    else if (t == 5) {
      buildBlockPose(
        poseTarget,
        1
      );
    }

    else if (t == 6) {
      buildBlockPose(
        poseTarget,
        2
      );
    }

    else if (t == 7) {
      buildBlockPose(
        poseTarget,
        3
      );
    }

    else if (t == 8) {
      buildBlockPose(
        poseTarget,
        3
      );
    }

    else {
      buildBlockPose(
        poseTarget,
        3
      );
    }
  }


  // ============================================================
  // HAND ATTACKS
  // ============================================================

  else if (c == 1) {

    if (t <= 3) {

      buildPunchPose(
        poseTarget,
        t % 2 == 0,
        t == 2
      );
    }

    else if (t <= 7) {

      buildBlockPose(
        poseTarget,
        t % 3
      );
    }

    else {

      buildBlockPose(
        poseTarget,
        2
      );
    }
  }


  // ============================================================
  // KICKS
  // ============================================================

  else if (c == 2) {

    buildKickPose(
      poseTarget,
      t
    );
  }


  // ============================================================
  // FOOTWORK
  // ============================================================

  else if (c == 3) {

    buildFootworkPose(
      poseTarget,
      t
    );
  }


  // ============================================================
  // KUMITE MOVEMENT
  // ============================================================

  else if (c == 4) {

    buildKumiteMovementPose(
      poseTarget,
      t
    );
  }


  // ============================================================
  // KUMITE TACTICS
  // ============================================================

  else if (c == 5) {

    buildTacticPose(
      poseTarget,
      t
    );
  }


  // ============================================================
  // KATA
  // ============================================================

  else if (c == 6) {

    buildKataStudyPose(
      poseTarget,
      t
    );
  }
}

    p[25].z +=
      0.70 * s;

    // Tsugi-ashi

    p[23].z +=
      0.40 * s;

      0.40 * s;
  }


  else if (type == 2) {

    // Yori-ashi

    p[23].x -=
      0.40 * s;

    p[24].x -=
      0.40 * s;
  }


  else if (type == 3) {

    // Tai sabaki

    p[23].z -=
      0.25 * s;

    p[24].z +=
      0.25 * s;

    p[11].z +=
      0.25 * s;

    p[12].z -=
      0.25 * s;
  }


  else if (type == 4) {

    p[25].z +=
      0.80 * s;
  }


  else if (type == 5) {

    p[27].z -=
      0.75 * s;

    p[28].z -=
      0.75 * s;
  }


  else if (type == 6) {

    p[25].x -=
      0.55 * s;

    p[26].x -=
      0.55 * s;
  }


  else if (type == 7) {

    p[25].x +=
      0.45 * s;

    p[26].x -=
      0.45 * s;
  }


  else if (type == 8) {

    p[23].z +=
      0.30 * s;

    p[24].z -=
      0.30 * s;
  }


  else {

    p[23].z +=
      0.20 * s;

    p[24].z +=
      0.20 * s;
  }


  syncFingers(p);

  syncFeet(p);
}


// ================================================================
// KUMITE MOVEMENT
// ================================================================

void buildKumiteMovementPose(
  PVector[] p,
  int type
) {

  buildKamaePose(p);

  float s =
    height / 12.0;


  if (type == 0) {

    // Kamae
  }


  else if (type == 1) {

    // Ma-ai

    p[23].z -=
      0.35 * s;

    p[24].z -=
      0.35 * s;
  }


  else if (type == 2) {

    // Chudan

    p[15].z +=
      0.35 * s;
  }


  else if (type == 3) {

    // Jodan

    p[15].y -=
      0.70 * s;

    p[15].z +=
      0.35 * s;
  }


  else if (type == 4) {

    // Entry

    p[23].z +=
      0.60 * s;

    p[24].z +=
      0.60 * s;

    p[15].z +=
      0.60 * s;
  }


  else if (type == 5) {

    // Exit

    p[23].z -=
      0.60 * s;

    p[24].z -=
      0.60 * s;
  }


  else if (type == 6) {

    // Feint

    p[11].z +=
      0.30 * s;

    p[12].z -=
      0.30 * s;
  }


  else if (type == 7) {

    // Counterattack

    buildPunchPose(
      p,
      true,
      false
    );
  }


  else if (type == 8) {

    // Interception

    p[15].z +=
      1.00 * s;

    p[16].z +=
      0.30 * s;
  }


  else {

    // Angle change

    p[23].x -=
      0.35 * s;

    p[24].x +=
      0.35 * s;
  }


  syncFingers(p);

  syncFeet(p);
}

    height / 12.0;


  // --------------------------------------------------------------
  // Each kata receives a characteristic STUDY posture.
  //
  // This does not claim to reproduce the complete kata.
  // --------------------------------------------------------------

  if (type == 0) {

    // Heian Shodan

    buildZenkutsuPose(p);

    buildBlockPose(
      p,
      1
    );
  }


  else if (type == 1) {

    // Heian Nidan

    buildZenkutsuPose(p);

    buildBlockPose(
      p,
      0
    );
  }


  else if (type == 2) {

    // Heian Sandan

    buildKibaDachiPose(p);

    buildBlockPose(
      p,
      2
    );
  }


  else if (type == 3) {

    // Heian Yondan

    buildZenkutsuPose(p);

    buildBlockPose(
      p,
      0
    );

    p[15].z +=
      0.35 * s;
  }


  else if (type == 4) {

    // Heian Godan

    buildZenkutsuPose(p);

    p[25].z +=
      0.30 * s;

    p[27].z +=
      0.30 * s;
  }


  else if (type == 5) {

    // Tekki Shodan

    buildKibaDachiPose(p);
  }


  else if (type == 6) {

    // Bassai Dai study

    buildZenkutsuPose(p);

    p[13].z +=
      0.40 * s;

    p[15].z +=
      0.70 * s;
  }


  else if (type == 7) {

    // Kanku Dai study

    buildZenkutsuPose(p);

    p[15].y -=
      0.50 * s;

    p[15].z +=
      0.40 * s;
  }


  else if (type == 8) {

    // Jion study

    buildZenkutsuPose(p);

    buildBlockPose(
      p,
      2
    );
  }


  else {

    // Enpi study

    buildKokutsuPose(p);

    p[15].z +=
      0.60 * s;
  }


  syncFingers(p);

  syncFeet(p);
}


// ================================================================
// COPY POSE
// ================================================================

void copyPose(
  PVector[] src,
  PVector[] dst
) {

  for (int i = 0;
       i < NUM_POINTS;
       i++) {

    dst[i].set(
      src[i]
    );
  }
}


// ================================================================
// START TECHNIQUE
// ================================================================

void startTechnique() {

  copyPose(
    points,
    poseStart
  );


  buildSelectedTechnique();


  transition = 0.0;

  walking = false;
}


// ================================================================
// START WALK
// ================================================================

void startWalking() {

  copyPose(
    points,
    poseStart
  );

  walking = true;

  walkPhase = 0;

  transition = 0.0;
}


// ================================================================
// TOUCH / MOUSE PRESSED
// ================================================================

void mousePressed() {

  downX = mouseX;

  downY = mouseY;


  // ============================================================
  // HOME
  // ============================================================

  if (screen == HOME_SCREEN) {

    float margin =
      width * 0.055;

    float gap =
      width * 0.025;

    float bw =
      (width - 2 * margin - gap) / 2.0;

    float bh =
      height * 0.085;


    for (int i = 0;
         i < 7;
         i++) {

      int col =
        i % 2;

      int row =
        i / 2;


      float x =
        margin +
        col * (bw + gap);

      float y =
        height * 0.20 +
        row * (bh + gap);


      if (inside(
        mouseX,
        mouseY,
        x,
        y,
        bw,
        bh
      )) {

        selectedCategory =
          i;

        screen =
          CATEGORY_SCREEN;

        return;
      }
    }
  }


  // ============================================================
  // CATEGORY
  // ============================================================

  else if (
    screen == CATEGORY_SCREEN
  ) {


    // HOME

    if (inside(
      mouseX,
      mouseY,
      10,
      10,
      width * 0.20,
      height * 0.055
    )) {

      screen =
        HOME_SCREEN;

      return;
    }


    float margin =
      width * 0.05;

    float gap =
      width * 0.025;

    float bw =
      (width - 2 * margin - gap) / 2.0;

    float bh =
      height * 0.073;


    for (int i = 0;
         i < 10;
         i++) {

      int col =
        i % 2;

      int row =
        i / 2;


      float x =
        margin +
        col * (bw + gap);

      float y =
        height * 0.12 +
        row * (bh + gap);


      if (inside(
        mouseX,
        mouseY,
        x,
        y,
        bw,
        bh
      )) {

        selectedTechnique =
          i;

        screen =
          TECHNIQUE_SCREEN;

        startTechnique();

        return;
      }
    }
  }


  // ============================================================
  // TECHNIQUE
  // ============================================================

  else if (
    screen == TECHNIQUE_SCREEN
  ) {


    // BACK

    if (inside(
      mouseX,
      mouseY,
      10,
      10,
      width * 0.18,
      height * 0.055
    )) {

      walking = false;

      screen =
        CATEGORY_SCREEN;

      return;
    }


    // REPLAY

    if (inside(
      mouseX,
      mouseY,
      width * 0.54,
      height * 0.60,
      width * 0.19,
      height * 0.065
    )) {

      startTechnique();

      return;
    }


    // WALK

    if (inside(
      mouseX,
      mouseY,
      width * 0.75,
      height * 0.60,
      width * 0.19,
      height * 0.065
    )) {

      startWalking();

      return;
    }


    // MODEL DRAG

    if (
      mouseY >
      height * 0.10 &&
      mouseX <
      width * 0.52
    ) {

      draggingModel =
        true;
    }
  }
}


// ================================================================
// TOUCH RELEASE
// ================================================================

void mouseReleased() {

  draggingModel =
    false;
}


// ================================================================
// ROTATION
// ================================================================

void mouseDragged() {

  if (
    screen ==
    TECHNIQUE_SCREEN &&
    draggingModel
  ) {

    rotY +=
      (mouseX - pmouseX) *
      0.01;

    rotX -=
      (mouseY - pmouseY) *
      0.01;


    rotX =
      constrain(
        rotX,
        -1.45,
        1.45
      );
  }
}


// ================================================================
// INSIDE BUTTON
// ================================================================

boolean inside(
  float mx,
  float my,
  float x,
  float y,
  float w,
  float h
) {

  return (
    mx >= x &&
    mx <= x + w &&
    my >= y &&
    my <= y + h
  );
}


// ================================================================
// FLOOR
// ================================================================

void drawFloor() {

  float s =
    height / 12.0;


  float floorY =
    6.35 * s;


  stroke(
    45,
    60,
    70
  );

  strokeWeight(2);


  for (
    float i = -1200;
    i <= 1200;
    i += 120
  ) {


    line(
      i,
      floorY,
      -1200,

      i,
      floorY,
      1200
    );


    line(
      -1200,
      floorY,
      i,

      1200,
      floorY,
      i
    );
  }
}


// ================================================================
// DRAW BONES
// ================================================================

void drawBones() {

  stroke(
    0,
    235,
    160
  );

  strokeWeight(8);


  for (int e = 0;
       e < connections.length;
       e++) {


    int a =
      connections[e][0];

    int b =
      connections[e][1];


    PVector p1 =
      points[a];

    PVector p2 =
      points[b];


    line(
      p1.x,
      p1.y,
      p1.z,

      p2.x,
      p2.y,
      p2.z
    );
  }
}


// ================================================================
// DRAW JOINTS
// ================================================================

void drawJoints() {

  for (int i = 0;
       i < NUM_POINTS;
       i++) {


    pushMatrix();


    translate(
      points[i].x,
      points[i].y,
      points[i].z
    );


    noStroke();


    boolean major =
      (
        i == 11 ||
        i == 12 ||
        i == 13 ||
        i == 14 ||
        i == 15 ||
        i == 16 ||
        i == 23 ||
        i == 24 ||
        i == 25 ||
        i == 26 ||
        i == 27 ||
        i == 28
      );


    if (major) {

      fill(
        255,
        75,
        70
      );

      sphere(14);
    }

    else {

      fill(
        255,
        180,
        60
      );

      sphere(9);
    }


    // ------------------------------------------------------------
    // LANDMARK NUMBER
    // ------------------------------------------------------------

    hint(
      DISABLE_DEPTH_TEST
    );


    fill(255);

    textAlign(
      CENTER,
      CENTER
    );

    textSize(18);


    text(
      "" + i,
      0,
      -22
    );


    hint(
      ENABLE_DEPTH_TEST
    );


    popMatrix();
  }
}
